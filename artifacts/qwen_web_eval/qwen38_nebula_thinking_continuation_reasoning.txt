function probeLive(p, i){
  const s = p.status;
  let sig = p.sig + 1.6*Math.sin(tick*0.2 + i*1.7);
  let temp = p.temp + 1.2*Math.sin(tick*0.15 + i*2.1);
  let batt = clamp(p.batt + 0.6*Math.sin(tick*0.08 + i), 0, 100);
  if(s==="offline"){ sig = -118; temp = 14.2; }
  return { sig:Math.round(sig*10)/10, temp:Math.round(temp*10)/10, batt:Math.round(batt) };
}

function sparkline(vals, color){
  const w=78, hh=26, min=Math.min.apply(null,vals), max=Math.max.apply(null,vals), rng=(max-min)||1;
  const pts = vals.map((v,i)=> (i/(vals.length-1)*w).toFixed(1)+","+(hh-2-((v-min)/rng)*(hh-4)).toFixed(1)).join(" ");
  return '<svg class="spark" viewBox="0 0 '+w+' '+hh+'" preserveAspectRatio="none" aria-hidden="true"><polyline points="'+pts+'" fill="none" stroke="'+color+'" stroke-width="1.6" stroke-linejoin="round" stroke-linecap="round"/></svg>';
}

const SERIES = [
  {key:"signal",  label:"Signal Strength", unit:"dBm", color:"var(--accent)", on:true},
  {key:"thermal", label:"Thermal Load",    unit:"%",   color:"var(--warn)",   on:true},

let range = LS.get("range","6H");
const RANGE_POINTS = {"1H":48,"6H":72,"24H":96,"7D":120};
const RANGE_SPAN   = {"1H":60,"6H":360,"24H":1440,"7D":10080}; // minutes

function renderChart(inst){
  const svg = inst===1?$("#chartSvg"):$("#chartSvg2");
  const wrap = inst===1?$("#chartWrap"):$("#chartWrap2");
  const tip = inst===1?$("#chartTip"):$("#chartTip2");
  const W = Math.max(320, wrap.clientWidth||720);
  const H = inst===1?280:280;
  const m = {t:14,r:14,b:26,l:40};
  const iw = W-m.l-m.r, ih = H-m.t-m.b;
  svg.setAttribute("viewBox","0 0 "+W+" "+H);
  svg.setAttribute("width",W); svg.setAttribute("height",H);
  while(svg.firstChild) svg.removeChild(svg.firstChild);

  const N = RANGE_POINTS[range];
  const span = RANGE_SPAN[range];
  const now = new Date();
  const tEnd = now.getTime();
  const tStart = tEnd - span*60000;

  // grid + y ticks (0..100 index)
  for(let i=0;i<=4;i++){
    const y = m.t + ih - (i/4)*ih;
    svg.appendChild(el("line",{class:"grid",x1:m.l,y1:y,x2:m.l+iw,y2:y}));
    const tx = el("text",{x:m.l-8,y:y+3,"text-anchor":"end"}); tx.textContent=(i*25)+"%"; svg.appendChild(tx);
  }
  // x ticks
  const xticks=5;
  for(let i=0;i<=xticks;i++){
    const x = m.l + (i/xticks)*iw;
    const d = new Date(tStart + (i/xticks)*span*60000);
    const lbl = range==="7D" ? (d.getUTCMonth()+1)+"/"+d.getUTCDate() : fmtUTCShort(d);
    const tx = el("text",{x:x,y:H-8,"text-anchor":"middle"}); tx.textContent=lbl; svg.appendChild(tx);
  }
  svg.appendChild(el("line",{class:"axis",x1:m.l,y1:m.t+iw*0+ih,x2:m.l+iw,y2:m.t+ih}));

  // series paths
  const data = {};
  SERIES.forEach(s=>{
    if(!s.on) return;
    const vals=[];
    for(let i=0;i<N;i++){
      const t = tick - (N-1-i);
      vals.push(seriesVal(s.key, t));
    }
    data[s.key]=vals;
    // normalize to 0..100 using a stable per-series domain
    let lo,hi;
    if(s.key==="signal"){ lo=-100; hi=-78; }
    else { lo=20; hi=100; }
    const pts = vals.map((v,i)=>{
      const x = m.l + (i/(N-1))*iw;
      const y = m.t + ih - clamp((v-lo)/(hi-lo),0,1)*ih;
      return x.toFixed(1)+","+y.toFixed(1);
    }).join(" ");
    svg.appendChild(el("polyline",{class:"line",points:pts,stroke:s.color}));
    // live point
    const last = vals[vals.length-1];
    const lx = m.l+iw, ly = m.t+ih-clamp((last-lo)/(hi-lo),0,1)*ih;
    svg.appendChild(el("circle",{class:"pt live-dot",cx:lx,cy:ly,r:3.5,fill:s.color}));
  });

  // hover layer
  const cross = el("line",{class:"cross",x1:0,y1:m.t,x2:0,y2:m.t+ih,visibility:"hidden"});
  svg.appendChild(cross);
  const dots = {};
  SERIES.forEach(s=>{ const c=el("circle",{class:"pt",r:4,fill:s.color,visibility:"hidden"}); svg.appendChild(c); dots[s.key]=c; });

  function showHover(clientX){
    const rect = svg.getBoundingClientRect();
    const px = (clientX-rect.left) * (W/rect.width);
    const idx = clamp(Math.round((px-m.l)/iw*(N-1)),0,N-1);
    const x = m.l + (idx/(N-1))*iw;
    cross.setAttribute("x1",x); cross.setAttribute("x2",x); cross.setAttribute("visibility","visible");
    let rows="";
    SERIES.forEach(s=>{
      if(!s.on){ dots[s.key].setAttribute("visibility","hidden"); return; }
      const v = data[s.key][idx];
      const lo = s.key==="signal"?-100:20, hi = s.key==="signal"?-78:100;
      const y = m.t+ih-clamp((v-lo)/(hi-lo),0,1)*ih;
      dots[s.key].setAttribute("cx",x); dots[s.key].setAttribute("cy",y); dots[s.key].setAttribute("visibility","visible");
      rows += '<div class="t-row"><span class="sw" style="background:'+s.color+'"></span>'+s.label+'<b>'+ (s.key==="signal"?v.toFixed(1):Math.round(v)) +' '+s.unit+'</b></div>';
    });
    const d = new Date(tStart + (idx/(N-1))*span*60000);
    tip.innerHTML = '<div class="t-time">'+fmtUTCShort(d)+' UTC</div>'+rows;
    tip.style.opacity="1";
    const tw = tip.offsetWidth;
    let left = (x/W)*rect.width + 14;
    if(left+tw > rect.width) left = (x/W)*rect.width - tw - 14;
    tip.style.left = left+"px";
    tip.style.top = (m.t+6)+"px";
  }
  function hideHover(){
    cross.setAttribute("visibility","hidden");
    SERIES.forEach(s=>dots[s.key].setAttribute("visibility","hidden"));
    tip.style.opacity="0";
  }
  svg.onmousemove = e=>showHover(e.clientX);
  svg.onmouseleave = hideHover;
  svg.ontouchmove = e=>{ if(e.touches[0]) showHover(e.touches[0].clientX); };
  svg.ontouchend = hideHover;
}

  const moon=el("circle",{class:"moon",r:4}); scene.appendChild(moon);
  const mt=el("text",{x:0,y:0}); mt.textContent="MOON"; scene.appendChild(mt);
  // relay

  svg.appendChild(scene);
  applyOrbitZoom(inst);
  // animate
  if(!reduced.matches){
    const start = performance.now();
    (function loop(now){
      const e=(now-start)/1000;
      const aM = e*0.5;
      const aR = e*0.12 + 2;
      const aA = e*0.06 + 0.6;
      moon.setAttribute("cx",cx+52*Math.cos(aM)); moon.setAttribute("cy",cy+52*Math.sin(aM));
      mt.setAttribute("x",cx+52*Math.cos(aM)+7); mt.setAttribute("y",cy+52*Math.sin(aM)-4);
      relay.setAttribute("cx",cx+132*Math.cos(aR)); relay.setAttribute("cy",cy+132*Math.sin(aR));
      const rx=150*Math.cos(aA), ry=88*Math.sin(aA), ph=-18*Math.PI/180;
      const ax=cx+rx*Math.cos(ph)-ry*Math.sin(ph), ay=cy+rx*Math.sin(ph)+ry*Math.cos(ph);
      ast.setAttribute("cx",ax); ast.setAttribute("cy",ay);
      at.setAttribute("x",ax+9); at.setAttribute("y",ay-6);
      requestAnimationFrame(loop);
    })(start);
  } else {
    moon.setAttribute("cx",cx+52); moon.setAttribute("cy",cy);
    mt.setAttribute("x",cx+59); mt.setAttribute("y",cy-4);
    relay.setAttribute("cx",cx-132); relay.setAttribute("cy",cy);
    const ph=-18*Math.PI/180, rx=150*Math.cos(0.6), ry=88*Math.sin(0.6);
    const ax=cx+rx*Math.cos(ph)-ry*Math.sin(ph), ay=cy+rx*Math.sin(ph)+ry*Math.cos(ph);
    ast.setAttribute("cx",ax); ast.setAttribute("cy",ay); at.setAttribute("x",ax+9); at.setAttribute("y",ay-6);
  }
}

function statusChip(s){
  return '<span class="status '+s+'"><span class="sd"></span>'+s.charAt(0).toUpperCase()+s.slice(1)+'</span>';
}
function battCell(v){
  const cls = v<20?"crit":v<50?"warn":"";
  return '<div class="batt"><span class="bar"><i class="'+cls+'" style="width:'+v+'%"></i></span><span>'+v+'%</span></div>';
}
function filteredProbes(){
  const q = filters.q.trim().toLowerCase();
  let rows = PROBES.map((p,i)=>{
    const lv = probeLive(p,i);
    return Object.assign({}, p, {sig:lv.sig, temp:lv.temp, batt:lv.batt, _i:i});
  });
  if(filters.status!=="all") rows = rows.filter(r=>r.status===filters.status);
  if(q) rows = rows.filter(r=> (r.name+" "+r.id+" "+r.model+" "+r.op).toLowerCase().includes(q));
  const k=filters.sortKey, d=filters.sortDir;
  rows.sort((a,b)=>{
    let av=a[k], bv=b[k];
    if(k==="contact"){ av=a.contact; bv=b.contact; }
    if(typeof av==="string") return av.localeCompare(bv)*d;
    return (av-bv)*d;
  });
  return rows;
}

  const tr = inst===1?$("#fleetHead"):$("#fleetHead2");

      th.setAttribute("aria-sort", filters.sortKey===c.k ? (filters.sortDir===1?"ascending":"descending") : "none");
      th.innerHTML = c.l+'<span class="arrow">'+(filters.sortKey===c.k?(filters.sortDir===1?"↑":"↓"):"")+'</span>';
      th.addEventListener("click", ()=>{
        if(filters.sortKey===c.k) filters.sortDir*=-1; else { filters.sortKey=c.k; filters.sortDir=1; }
        LS.set("filters",filters); renderFleet();
      });
    } else th.textContent=c.l;
    tr.appendChild(th);
  });
}

  const kb = $("#kanban"); kb.innerHTML="";

      elc.addEventListener("dragstart", e=>{ dragId=card.id; elc.classList.add("dragging"); e.dataTransfer.setData("text/plain",card.id); e.dataTransfer.effectAllowed="move"; });
      elc.addEventListener("dragend", ()=>{ elc.classList.remove("dragging"); dragId=null; });

        e.stopPropagation(); openMoveMenu(elc, card.id);
      });

      e.preventDefault(); c.classList.remove("drag-over");
      const id = e.dataTransfer.getData("text/plain")||dragId;
      if(id) moveCard(id, col.id);
    });
    kb.appendChild(c);
  });
}
function moveCard(id, toCol){
  for(const k in board){ const i=board[k].findIndex(c=>c.id===id); if(i>-1){ const [c]=board[k].splice(i,1); if(k!==toCol) board[toCol].push(c); } }
  LS.set("kanban", board);
  renderKanban();
  toast("Task moved to "+KANBAN_COLS.find(c=>c.id===toCol).label, "info");
}
function openMoveMenu(anchor, id){
  $$(".kb-menu").forEach(m=>m.remove());
  const menu = h("div",{class:"kb-menu",role:"menu"});
  KANBAN_COLS.forEach(col=>{
    const cur = (board[col.id]||[]).some(c=>c.id===id);
    const b=h("button",{role:"menuitem","aria-current":String(cur)});
    b.textContent = col.label;
    b.addEventListener("click", ()=>{ moveCard(id,col.id); menu.remove(); });
    menu.appendChild(b);
  });
  anchor.appendChild(menu);
  setTimeout(()=>document.addEventListener("click", function h(e){ if(!menu.contains(e.target)) menu.remove(); document.removeEventListener("click",h); }),0);
}

/* ============================================================
   CLOCKS
   ============================================================ */
function tickClocks(){
  const now=new Date();
  $("#utcClock").textContent=fmtUTC(now)+" UTC";
  $("#utcClock").setAttribute("datetime", now.toISOString());
  const ms = now.getTime()-MISSION_START;
  const d=Math.floor(ms/86400000), hr=Math.floor(ms%86400000/3600000), mn=Math.floor(ms%3600000/60000), sc=Math.floor(ms%60000/1000);
  $("#missionClock").textContent=d+"d "+pad(hr)+":"+pad(mn)+":"+pad(sc);
}
