  function renderSparklines() {

  function drawSparkline(canvasId, data, color) {
    const canvas = $(canvasId);
    if (!canvas) return;
    const ctx = canvas.getContext('2d');
    const dpr = window.devicePixelRatio || 1;
    const rect = canvas.getBoundingClientRect();
    canvas.width = rect.width * dpr;
    canvas.height = rect.height * dpr;
    ctx.scale(dpr, dpr);
    const w = rect.width;
    const h = rect.height;
    const min = Math.min(...data);
    const max = Math.max(...data);
    const range = max - min || 1;
    const step = w / (data.length - 1);

    ctx.clearRect(0, 0, w, h);
    ctx.beginPath();
    ctx.strokeStyle = color;
    ctx.lineWidth = 1.5;
    ctx.lineJoin = 'round';
    data.forEach((v, i) => {
      const x = i * step;
      const y = h - ((v - min) / range) * (h - 4) - 2;
      if (i === 0) ctx.moveTo(x, y);
      else ctx.lineTo(x, y);
    });
    ctx.stroke();

    // Fill
    ctx.lineTo(w, h);
    ctx.lineTo(0, h);
    ctx.closePath();
    ctx.fillStyle = color.replace(')', ',0.1)').replace('rgb', 'rgba');
    ctx.fill();
  }

  function renderTelemetryChart() {
    const canvas = $('telemetryCanvas');
    if (!canvas) return;
    const ctx = canvas.getContext('2d');
    const dpr = window.devicePixelRatio || 1;
    const rect = canvas.parentElement.getBoundingClientRect();
    canvas.width = rect.width * dpr;
    canvas.height = rect.height * dpr;
    canvas.style.width = rect.width + 'px';
    canvas.style.height = rect.height + 'px';
    ctx.scale(dpr, dpr);

    const w = rect.width;
    const h = rect.height;
    const pad = { top: 20, right: 20, bottom: 35, left: 50 };
    const cw = w - pad.left - pad.right;
    const ch = h - pad.top - pad.bottom;

    ctx.clearRect(0, 0, w, h);

    // Grid
    ctx.strokeStyle = getComputedStyle(document.documentElement).getPropertyValue('--color-border').trim();
    ctx.lineWidth = 0.5;
    for (let i = 0; i <= 5; i++) {
      const y = pad.top + (ch / 5) * i;
      ctx.beginPath();
      ctx.moveTo(pad.left, y);
      ctx.lineTo(w - pad.right, y);
      ctx.stroke();
    }

    // Y-axis labels
    ctx.fillStyle = getComputedStyle(document.documentElement).getPropertyValue('--color-text-muted').trim();
    ctx.font = '10px ' + getComputedStyle(document.documentElement).getPropertyValue('--font-mono').trim();
    ctx.textAlign = 'right';
    for (let i = 0; i <= 5; i++) {
      const y = pad.top + (ch / 5) * i;
      const val = 100 - i * 20;
      ctx.fillText(val, pad.left - 8, y + 4);
    }

    // X-axis labels
    ctx.textAlign = 'center';
    const timeLabels = ['00:00', '04:00', '08:00', '12:00', '16:00', '20:00', '24:00'];
    timeLabels.forEach((label, i) => {
      const x = pad.left + (cw / (timeLabels.length - 1)) * i;
      ctx.fillText(label, x, h - pad.bottom + 18);
    });

    // Draw series
    const seriesColors = { signal: '#06b6d4', thermal: '#f59e0b', power: '#10b981' };
    const seriesKeys = Object.keys(state.telemetryVisible).filter(k => state.telemetryVisible[k]);

    seriesKeys.forEach(key => {
      const data = telemetryData[key];
      const color = seriesColors[key];
      const min = Math.min(...data);
      const max = Math.max(...data);
      const range = max - min || 1;
      const step = cw / (data.length - 1);

      ctx.beginPath();
      ctx.strokeStyle = color;
      ctx.lineWidth = 2;
      ctx.lineJoin = 'round';
      data.forEach((v, i) => {
        const x = pad.left + i * step;
        const y = pad.top + ch - ((v - min) / range) * ch;
        if (i === 0) ctx.moveTo(x, y);
        else ctx.lineTo(x, y);
      });
      ctx.stroke();

      // Current point highlight
      const lastX = pad.left + (data.length - 1) * step;
      const lastY = pad.top + ch - ((data[data.length - 1] - min) / range) * ch;
      ctx.beginPath();
      ctx.arc(lastX, lastY, 4, 0, Math.PI * 2);
      ctx.fillStyle = color;
      ctx.fill();
      ctx.beginPath();
      ctx.arc(lastX, lastY, 7, 0, Math.PI * 2);
      ctx.strokeStyle = color;
      ctx.lineWidth = 1;
      ctx.globalAlpha = 0.3;
      ctx.stroke();
      ctx.globalAlpha = 1;
    });

    // Store chart info for tooltip
    canvas._chartInfo = { pad, cw, ch, w, h };
  }

    const moonY = cy + Math.sin(moonAngle) * moonR;

    ctx.ellipse(cx, cy, 80 * zoom, 55 * zoom, -0.2, 0, Math.PI * 2);

      const py = cy + Math.cos(angle) * rx * Math.sin(tilt) + Math.sin(angle) * ry * Math.cos(tilt);
      ctx.beginPath();
      ctx.arc(px, py, 3 * zoom, 0, Math.PI * 2);
      ctx.fillStyle = accent;
      ctx.fill();
    });
  }

  // ===== FLEET TABLE =====
  function getFilteredFleet() {
    let data = [...fleetData];
    if (state.tableFilter !== 'all') {
      data = data.filter(p => p.status === state.tableFilter);
    }
    if (state.tableSearch) {
      const q = state.tableSearch.toLowerCase();
      data = data.filter(p => p.name.toLowerCase().includes(q) || p.model.toLowerCase().includes(q) || p.operator.toLowerCase().includes(q));
    }
    // Sort
    const key = state.tableSort.key;
    const dir = state.tableSort.dir === 'asc' ? 1 : -1;
    data.sort((a, b) => {
      let va = a[key], vb = b[key];
      if (key === 'distance' || key === 'signal' || key === 'temp' || key === 'power') {
        va = parseFloat(va);
        vb = parseFloat(vb);
      }
      if (typeof va === 'string') {
        return va.localeCompare(vb) * dir;
      }
      return (va - vb) * dir;
    });
    return data;
  }

  function renderKanban() {

      `).join('');

  function setupKanbanDrag() {
    const cards = $$('.kanban-card');
    const zones = $$('.kanban-drop-zone');

    cards.forEach(card => {
      card.addEventListener('dragstart', e => {
        e.dataTransfer.setData('text/plain', JSON.stringify({ id: card.dataset.id, column: card.dataset.column }));
        card.classList.add('dragging');
      });
      card.addEventListener('dragend', () => card.classList.remove('dragging'));
    });

    zones.forEach(zone => {
      zone.addEventListener('dragover', e => { e.preventDefault(); zone.classList.add('drag-over'); });
      zone.addEventListener('dragleave', () => zone.classList.remove('drag-over'));
      zone.addEventListener('drop', e => {
        e.preventDefault();
        zone.classList.remove('drag-over');
        const data = JSON.parse(e.dataTransfer.getData('text/plain'));
        const targetCol = zone.id.replace('zone', '').toLowerCase();
        const colMap = { planning: 'planning', transit: 'in-transit', orbit: 'orbit', complete: 'complete' };
        const target = colMap[targetCol];
        if (!target || target === data.column) return;

        // Remove from source
        const srcItems = state.kanban[data.column];
        const idx = srcItems.findIndex(x => x.id === data.id);
        if (idx > -1) {
          const [item] = srcItems.splice(idx, 1);
          state.kanban[target].push(item);
          renderKanban();
          showToast('info', `Moved "${item.title}" to ${capitalize(target.replace('-', ' '))}`);
        }
      });
    });
  }

  function setupKanbanMobile() {
    $$('.move-menu-btn').forEach(btn => {
      btn.addEventListener('click', e => {
        e.stopPropagation();
        const menu = $('moveMenu');
        const rect = btn.getBoundingClientRect();
        menu.style.top = rect.bottom + 'px';
        menu.style.left = rect.left + 'px';
        menu.innerHTML = ['planning', 'in-transit', 'orbit', 'complete'].filter(c => c !== btn.dataset.column).map(c =>
          `<div class="move-menu-item" data-target="${c}">${capitalize(c.replace('-', ' '))}</div>`
        ).join('');
        menu.classList.add('open');
        menu.querySelectorAll('.move-menu-item').forEach(item => {
          item.addEventListener('click', () => {
            const target = item.dataset.target;
            const srcCol = btn.dataset.column;
            const srcItems = state.kanban[srcCol];
            const idx = srcItems.findIndex(x => x.id === btn.dataset.id);
            if (idx > -1) {
              const [moved] = srcItems.splice(idx, 1);
              state.kanban[target].push(moved);
              renderKanban();
              showToast('info', `Moved "${moved.title}" to ${capitalize(target.replace('-', ' '))}`);
            }
            menu.classList.remove('open');
          });
        });
      });
    });
  }
